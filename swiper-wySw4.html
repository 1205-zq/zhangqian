核心配置

const zq_wySw4 = new Swiper('#wySw4', {
  effect: 'cube',            // ④ 初始效果为 cube
  loop: true,               // ① 无缝循环
  autoplay: {
    delay: 3200,             // ② 3.2 秒自动翻页
    disableOnInteraction: false // ② 手动操作后不停止
  },
  pagination: {
    el: '.zq_pagination',
    clickable: true          // ③ 圆点分页器可点击
  },
  cubeEffect: { shadow: true, slideShadows: true, shadowOffset: 18, shadowScale: 0.94 },
  speed: 700,
  grabCursor: true
});
